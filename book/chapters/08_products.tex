\chapter{Product Analysis / Produktanalyse / 产品分析}

\section{Product Portfolio / Produktportfolio / 产品组合}

\begin{otherlanguage}{ngerman}
Edekas Produktangebot umfasst:
\begin{itemize}
    \item \textbf{Frischwaren}: Obst, Gemüse, Fleisch, Fisch, Backwaren
    \item \textbf{Trockenwaren}: Konserven, Nudeln, Getreide
    \item \textbf{Getränke}: Alkoholische und nicht-alkoholische Getränke
    \item \textbf{Non-Food}: Drogerie, Haushaltswaren, Textilien
    \item \textbf{Eigenmarken}: Gut \& Günstig, Edeka Bio, Edeka Feine Welt
\end{itemize}
\end{otherlanguage}

Edeka's product offering includes:
\begin{itemize}
    \item \textbf{Fresh goods}: Fruit, vegetables, meat, fish, baked goods
    \item \textbf{Dry goods}: Canned goods, pasta, grains
    \item \textbf{Beverages}: Alcoholic and non-alcoholic drinks
    \item \textbf{Non-food}: Drugstore, household goods, textiles
    \item \textbf{Private labels}: Gut \& Günstig, Edeka Bio, Edeka Feine Welt
\end{itemize}

艾德卡的产品包括：
\begin{itemize}
    \item \textbf{生鲜}：水果、蔬菜、肉类、鱼类、烘焙食品
    \item \textbf{干货}：罐头、意大利面、谷物
    \item \textbf{饮料}：含酒精和不含酒精的饮料
    \item \textbf{非食品}：药妆、家居用品、纺织品
    \item \textbf{自有品牌}：Gut \& Günstig、Edeka Bio、Edeka Feine Welt
\end{itemize}
